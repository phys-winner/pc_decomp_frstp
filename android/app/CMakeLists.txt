cmake_minimum_required(VERSION 3.22)
project(frostpunk_native_window_probe C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-g0 -ffunction-sections -fdata-sections)
if(NOT FP_GPU_SYSTEM)
    add_library(adrenotools SHARED IMPORTED)
    set_target_properties(adrenotools PROPERTIES IMPORTED_LOCATION "${GPU_RUNTIME}/libadrenotools.so")
endif()
if(FP_GAME_RUNTIME AND FP_DXVK_FIXTURE)
    message(FATAL_ERROR "Game worker and diagnostic renderer are exclusive")
endif()
if(FP_GAME_RUNTIME)
    set(PROBE_SOURCE)
elseif(FP_DXVK_FIXTURE)
    set(PROBE_SOURCE dxvk_probe.cpp "${FIXTURE_GENERATED}/guest_present_lifted.c")
else()
    set(PROBE_SOURCE window_probe.c)
endif()
add_library(frostpunk_app SHARED app.c ${PROBE_SOURCE} ../gpu/driver.c
    ../host/zip_files.c ../../reference/launcher-zipmount/zipmount.c
    "${ANDROID_NDK}/sources/android/native_app_glue/android_native_app_glue.c")
target_compile_definitions(frostpunk_app PRIVATE VK_NO_PROTOTYPES VK_USE_PLATFORM_ANDROID_KHR)
if(FP_GPU_SYSTEM)
    target_compile_definitions(frostpunk_app PRIVATE FP_GPU_SYSTEM=1)
endif()
target_include_directories(frostpunk_app PRIVATE "${ANDROID_NDK}/sources/android/native_app_glue"
    "${CMAKE_CURRENT_SOURCE_DIR}/../../reference/adrenotools/include")
target_link_libraries(frostpunk_app PRIVATE android log dl m)
if(NOT FP_GPU_SYSTEM)
    target_link_libraries(frostpunk_app PRIVATE adrenotools)
endif()
if(FP_GAME_RUNTIME)
    target_link_libraries(frostpunk_app PRIVATE nativewindow)
    target_compile_definitions(frostpunk_app PRIVATE FP_GAME_RUNTIME)
    string(LENGTH "${FP_ORIGINAL_EXE_SHA256}" FP_HASH_LENGTH)
    if(NOT FP_HASH_LENGTH EQUAL 64 OR NOT FP_ORIGINAL_EXE_SHA256 MATCHES "^[0-9a-f]+$")
        message(FATAL_ERROR "Game worker requires a pinned original executable SHA-256")
    endif()
    target_compile_definitions(frostpunk_app PRIVATE FP_ORIGINAL_EXE_SHA256="${FP_ORIGINAL_EXE_SHA256}")
    target_link_options(frostpunk_app PRIVATE -Wl,-z,global)
endif()
if(FP_DXVK_FIXTURE)
    target_compile_definitions(frostpunk_app PRIVATE FP_DXVK_FIXTURE)
    if(FP_NATIVE_PRESENT)
        target_compile_definitions(frostpunk_app PRIVATE FP_NATIVE_PRESENT)
    endif()
    target_compile_options(frostpunk_app PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wno-extern-c-compat>)
    target_include_directories(frostpunk_app PRIVATE
        "${CMAKE_CURRENT_SOURCE_DIR}/../../native/runtime64"
        "${CMAKE_CURRENT_SOURCE_DIR}/../../reference/dxvk/include/native/windows"
        "${CMAKE_CURRENT_SOURCE_DIR}/../../reference/dxvk/include/native/directx")
    target_link_libraries(frostpunk_app PRIVATE "${DXVK_BRIDGE}/libwin64_d3d11.a")
    # DXVK's explicitly required weak provider resolves from this embedding DSO.
    target_link_options(frostpunk_app PRIVATE -Wl,-z,global)
endif()
target_link_options(frostpunk_app PRIVATE -Wl,--gc-sections -Wl,--icf=safe -Wl,--pack-dyn-relocs=android+relr
    -Wl,--no-undefined -Wl,-u,ANativeActivity_onCreate -Wl,-z,max-page-size=16384)
