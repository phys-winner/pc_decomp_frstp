cmake_minimum_required(VERSION 3.22)
project(frostpunk_gpu_runtime LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
add_subdirectory(../../reference/adrenotools ${CMAKE_CURRENT_BINARY_DIR}/adrenotools)
add_library(frostpunk_gpu STATIC driver.c)
target_include_directories(frostpunk_gpu PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(frostpunk_gpu PUBLIC adrenotools dl)
add_executable(frostpunk_gpu_probe gpu_probe.c)
target_include_directories(frostpunk_gpu_probe PRIVATE ${GPU_GENERATED_DIR})
target_link_libraries(frostpunk_gpu_probe PRIVATE frostpunk_gpu)
target_compile_options(frostpunk_gpu PRIVATE -Wall -Wextra -Werror)
target_compile_options(frostpunk_gpu_probe PRIVATE -Wall -Wextra -Werror)
