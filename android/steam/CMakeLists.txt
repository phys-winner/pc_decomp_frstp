cmake_minimum_required(VERSION 3.22)
project(frostpunk_offline_services LANGUAGES C CXX)
set(DLC_CONFIG "${CMAKE_CURRENT_SOURCE_DIR}/../../config/DLC.txt")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${DLC_CONFIG}")
file(READ "${DLC_CONFIG}" FROSTPUNK_DLC_CONFIG)
configure_file(dlc_config.h.in "${CMAKE_CURRENT_BINARY_DIR}/dlc_config.h" @ONLY)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(protobuf_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(protobuf_BUILD_PROTOC_BINARIES OFF CACHE BOOL "" FORCE)
set(protobuf_BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(protobuf_WITH_ZLIB OFF CACHE BOOL "" FORCE)
set(protobuf_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory("${PROTOBUF_SOURCE}" "${CMAKE_BINARY_DIR}/protobuf" EXCLUDE_FROM_ALL)
file(GLOB GOLDBERG_CPP CONFIGURE_DEPENDS "${GOLDBERG_SOURCE}/dll/*.cpp")
# glibc LD_PRELOAD wrappers (__xstat64 et al.) have no Android ABI or purpose.
list(FILTER GOLDBERG_CPP EXCLUDE REGEX "/wrap\\.cpp$")
add_library(frostpunk_offline_services SHARED ${GOLDBERG_CPP}
    "${GENERATED_DIR}/net.pb.cc" offline_policy.cpp)
target_include_directories(frostpunk_offline_services PRIVATE "${GOLDBERG_SOURCE}" "${GENERATED_DIR}" "${PROTOBUF_SOURCE}/src" "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_definitions(frostpunk_offline_services PRIVATE EMU_RELEASE_BUILD)
target_compile_options(frostpunk_offline_services PRIVATE -Wno-return-type -Wno-inconsistent-missing-override -Wno-switch)
target_link_libraries(frostpunk_offline_services PRIVATE libprotobuf-lite dl)
target_link_options(frostpunk_offline_services PRIVATE -Wl,--wrap=socket -Wl,--no-undefined -Wl,--exclude-libs,ALL -Wl,-z,max-page-size=16384)
add_executable(offline_services_probe probe.cpp)
target_include_directories(offline_services_probe PRIVATE "${GOLDBERG_SOURCE}/sdk_includes")
target_link_libraries(offline_services_probe PRIVATE dl)
target_link_options(offline_services_probe PRIVATE -Wl,-z,max-page-size=16384)
