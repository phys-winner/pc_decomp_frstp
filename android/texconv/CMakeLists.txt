cmake_minimum_required(VERSION 3.22)
project(frostpunk_texconv LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
file(GLOB ASTC_SOURCES ../../reference/astcenc/Source/astcenc_*.cpp)
add_library(fp_astc STATIC ${ASTC_SOURCES})
target_compile_definitions(fp_astc PRIVATE ASTCENC_NEON=1 ASTCENC_SVE=0 ASTCENC_SSE=0 ASTCENC_AVX=0 ASTCENC_POPCNT=0 ASTCENC_F16C=0)
target_compile_options(fp_astc PRIVATE -O2 -fvisibility=hidden -ffunction-sections -fdata-sections)
add_library(frostpunk_texconv SHARED texconv.cpp)
target_compile_options(frostpunk_texconv PRIVATE -O2 -Wall -Wextra -Werror -fvisibility=hidden)
target_link_libraries(frostpunk_texconv PRIVATE fp_astc)
target_link_options(frostpunk_texconv PRIVATE -Wl,--no-undefined -Wl,--gc-sections -Wl,-z,max-page-size=16384)
option(FP_TEXTURE_TEST "Build the texture conversion self-check" OFF)
if(FP_TEXTURE_TEST)
  add_executable(texture_conversion_test texconv.cpp)
  target_compile_definitions(texture_conversion_test PRIVATE FP_TEXTURE_TEST=1)
  target_link_libraries(texture_conversion_test PRIVATE fp_astc)
endif()
