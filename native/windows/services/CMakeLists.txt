cmake_minimum_required(VERSION 3.22)
project(frostpunk_windows_services LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(protobuf_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(protobuf_BUILD_PROTOC_BINARIES OFF CACHE BOOL "" FORCE)
set(protobuf_BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(protobuf_WITH_ZLIB OFF CACHE BOOL "" FORCE)
set(protobuf_INSTALL OFF CACHE BOOL "" FORCE)
set(protobuf_MSVC_STATIC_RUNTIME ON CACHE BOOL "" FORCE)
add_subdirectory("${PROTOBUF_SOURCE}" "${CMAKE_BINARY_DIR}/protobuf" EXCLUDE_FROM_ALL)
file(GLOB GOLDBERG_CPP CONFIGURE_DEPENDS "${GOLDBERG_SOURCE}/dll/*.cpp")
list(FILTER GOLDBERG_CPP EXCLUDE REGEX "/wrap\\.cpp$")
add_custom_command(OUTPUT "${CMAKE_BINARY_DIR}/rtlgenrandom.lib"
    COMMAND "${CMAKE_AR}" /nologo /machine:x64 "/def:${GOLDBERG_SOURCE}/dll/rtlgenrandom.def"
        "/out:${CMAKE_BINARY_DIR}/rtlgenrandom.lib"
    DEPENDS "${GOLDBERG_SOURCE}/dll/rtlgenrandom.def")
add_custom_target(rtlgenrandom_import DEPENDS "${CMAKE_BINARY_DIR}/rtlgenrandom.lib")
add_library(steam_api64 SHARED ${GOLDBERG_CPP} "${GENERATED_DIR}/net.pb.cc")
add_dependencies(steam_api64 rtlgenrandom_import)
target_include_directories(steam_api64 PRIVATE "${GOLDBERG_SOURCE}" "${GENERATED_DIR}" "${PROTOBUF_SOURCE}/src")
target_compile_definitions(steam_api64 PRIVATE EMU_RELEASE_BUILD _CRT_SECURE_NO_WARNINGS)
target_compile_options(steam_api64 PRIVATE /EHsc /W0)
target_link_libraries(steam_api64 PRIVATE libprotobuf-lite "${CMAKE_BINARY_DIR}/rtlgenrandom.lib"
    Iphlpapi Ws2_32 Shell32 Advapi32)
add_executable(windows_services_probe "${CMAKE_CURRENT_SOURCE_DIR}/../../../tests/windows_services_probe.cpp")
target_include_directories(windows_services_probe PRIVATE "${GOLDBERG_SOURCE}/sdk_includes")
target_compile_definitions(windows_services_probe PRIVATE STEAM_WIN32)
target_compile_options(windows_services_probe PRIVATE /EHsc /W3 /UNDEBUG)
